  ```typescript
      action: "confirm",
    };
  ```

  **handleCounter function:**
  ```typescript
  const handleCounter = () => {
    if (!counterTime.trim()) return;
    const msg: AppointmentMessage = {
      type: "confirm_appointment",
      thread_id: props.thread_id,
      proposed_time: props.proposed_appointment ?? "",
      action: "counter",
      counter_time: counterTime.trim(),
    };
    props.send(JSON.stringify(msg));
    setCounterTime("");
    setShowCounter(false);
  };
  ```

  **Appointments section JSX (add below existing escalation section):**
  ```tsx
  {props.stage === "propose_appointment" && props.proposed_appointment && (
    <section className="appointments-section" aria-label="Appointments">
      <h3>Appointment Proposal</h3>
      <p className="proposed-time">{props.proposed_appointment}</p>
      <div className="appointment-actions">
        <button onClick={handleConfirm}>Confirm</button>
        <button onClick={() => setShowCounter(!showCounter)}>Counter-propose</button>
      </div>
      {showCounter && (
        <div className="counter-propose">
          <input
            type="text"
            value={counterTime}
            onChange={(e) => setCounterTime(e.target.value)}
            placeholder="e.g. Biyernes ng June 21 sa ika-3 ng hapon"
            aria-label="Counter-propose time"
          />
          <button onClick={handleCounter}>Send Counter</button>
        </div>
      )}
    </section>
  )}
  ```
